## 1. 專案概述

我們將構建 **TaskAPI**——一個包含以下端點的極簡任務管理器 REST API：

| 方法 | 路徑 | 描述 |
|---|---|---|
| `GET` | `/tasks` | 列出所有任務 |
| `POST` | `/tasks` | 建立任務 |
| `GET` | `/tasks/:id` | 取得單個任務 |
| `PATCH` | `/tasks/:id` | 更新完成狀態 |
| `DELETE` | `/tasks/:id` | 刪除任務 |

這與你用 Express + 記憶體儲存構建的內容非常接近，因此你可以專注於 Swift/Vapor 的差異。

```swift !! swift
// Swift：Codable 結構體——免費獲得序列化！

// Sources/TaskAPI/Models/Task.swift
import Vapor
import Foundation

// Content = Vapor 的請求/回應型別協定（封裝了 Codable）
struct Task: Content, Identifiable {
    let id: UUID
    var title: String
    var completed: Bool
    let createdAt: Date

    init(title: String) throws {
        guard !title.trimmingCharacters(in: .whitespaces).isEmpty else {
            throw Abort(.badRequest, reason: "Title cannot be empty")
        }
        self.id = UUID()
        self.title = title.trimmingCharacters(in: .whitespaces)
        self.completed = false
        self.createdAt = Date()
    }
}

// 建立任務的 DTO（只包含客戶端傳送的內容）
struct CreateTaskRequest: Content {
    let title: String
}

// 更新任務的 DTO
struct UpdateTaskRequest: Content {
    let completed: Bool?
    let title: String?
}

// 記憶體儲存（使用 actor 保證執行緒安全——見第 9 模組）
actor TaskStore {
    private var tasks: [UUID: Task] = [:]

    func all() -> [Task] { Array(tasks.values).sorted { $0.createdAt < $1.createdAt } }
    func get(_ id: UUID) -> Task? { tasks[id] }
    func create(_ task: Task) { tasks[task.id] = task }
    func update(_ task: Task) { tasks[task.id] = task }
    func delete(_ id: UUID) { tasks.removeValue(forKey: id) }
}
```

## 4. Vapor 中的路由

<UniversalEditor title="路由：Express 與 Vapor 對比" compare={true}>
```javascript !! js
// Express 路由
const router = Router();

// GET /tasks
router.get('/', (req, res) => {
  const tasks = Array.from(taskStore.values());
  res.json(tasks);
});

// POST /tasks
router.post('/', (req, res) => {
  try {
    const task = new Task(req.body.title);
    taskStore.set(task.id, task);
    res.status(201).json(task);
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});

// GET /tasks/:id
router.get('/:id', (req, res) => {
  const task = taskStore.get(req.params.id);
  if (!task) return res.status(404).json({ error: 'Task not found' });
  res.json(task);
});

// DELETE /tasks/:id
router.delete('/:id', (req, res) => {
  if (!taskStore.delete(req.params.id)) {
    return res.status(404).json({ error: 'Task not found' });
  }
  res.status(204).send();
});

export default router;
```

```swift !! swift
// Vapor 路由
import Vapor

let store = TaskStore()

func routes(_ app: Application) throws {
    let tasks = app.grouped("tasks")

    // GET /tasks
    tasks.get { req async throws -> [Task] in
        await store.all()
    }

    // POST /tasks
    tasks.post { req async throws -> Response in
        let body = try req.content.decode(CreateTaskRequest.self)
        let task = try Task(title: body.title)
        await store.create(task)
        return try await task.encodeResponse(status: .created, for: req)
    }

    // GET /tasks/:id
    tasks.get(":id") { req async throws -> Task in
        let id = try req.parameters.require("id", as: UUID.self)
        guard let task = await store.get(id) else {
            throw Abort(.notFound, reason: "Task \(id) not found")
        }
        return task
    }

    // PATCH /tasks/:id
    tasks.patch(":id") { req async throws -> Task in
        let id = try req.parameters.require("id", as: UUID.self)
        guard var task = await store.get(id) else {
            throw Abort(.notFound)
        }
        let update = try req.content.decode(UpdateTaskRequest.self)
        if let completed = update.completed { task.completed = completed }
        if let title = update.title { task.title = title }
        await store.update(task)
        return task
    }

    // DELETE /tasks/:id
    tasks.delete(":id") { req async throws -> HTTPStatus in
        let id = try req.parameters.require("id", as: UUID.self)
        guard await store.get(id) != nil else { throw Abort(.notFound) }
        await store.delete(id)
        return .noContent
    }
}
```

```javascript !! js
app.get('/tasks', (req, res) => {
```

```swift !! swift
// Swift：Codable 自動處理一切

// 解碼傳入的 JSON
app.post("tasks") { req async throws -> Task in
    let body = try req.content.decode(CreateTaskRequest.self)
    return try Task(title: body.title)
}

// 編碼為 JSON — 只需回傳遵循 Content 的型別
app.get("tasks") { req async throws -> [Task] in
    return await store.all()
}

// 自訂 DTO
struct TaskDTO: Content {
    let taskId: UUID
    let name: String
    let isDone: Bool
    let createdAt: String

    init(from task: Task) {
        self.taskId = task.id
        self.name = task.title
        self.isDone = task.completed
        self.createdAt = ISO8601DateFormatter().string(from: task.createdAt)
    }
}

// 帶泛型的分頁回應物件
struct PagedResponse<T: Content>: Content {
    let data: [T]
    let total: Int
    let page: Int
}

app.get("tasks") { req async throws -> PagedResponse<TaskDTO> in
    let tasks = await store.all()
    return PagedResponse(data: tasks.map(TaskDTO.init), total: tasks.count, page: 1)
```

```swift !! swift
import Vapor

// 非同步中介軟體
struct AuthMiddleware: AsyncMiddleware {
    func respond(to request: Request, chainingTo next: AsyncResponder) async throws -> Response {
        guard let token = request.headers.bearerAuthorization?.token else {
            throw Abort(.unauthorized, reason: "Missing bearer token")
```

```swift !! swift
// Swift 陷阱：過度使用 !（強制解包）

// 錯誤——如果值為 nil 會在執行時崩潰

let port = Int(ProcessInfo.processInfo.environment["PORT"]!)!

// 正確——使用 guard let 或 if let

func getConfig(_ key: String) throws -> String {
```

```swift !! swift
// Swift 陷阱：結構體是值型別——變異建立副本

struct Task {
    var completed: Bool
}

// 錯誤：函式接收副本；原始值不會被變異
func completeTask(_ task: Task) {
    var copy = task
    copy.completed = true   // 修改副本，不修改原始值

var myTask = Task(completed: false)

// 正確方案 1：inout 參數
func completeTaskInout(_ task: inout Task) {

}
completeTaskInout(&myTask)
print(myTask.completed)  // true

// 正確方案 2：回傳修改後的副本
func completed(_ task: Task) -> Task {
    var copy = task
    copy.completed = true
    return copy
}
myTask = completed(myTask)
```

```swift !! swift
// Swift 慣用法：guard let 提前回傳（無厄運金字塔）
func processUser(_ user: User?) throws -> ProcessResult {
    guard let user = user else { throw UserError.noUser }
    guard let profile = user.profile else { throw UserError.noProfile }
    guard let email = profile.email else { throw UserError.noEmail }
    guard isValidEmail(email) else { throw UserError.invalidEmail(email) }

    // 到這裡，所有值都已安全解包且有效
    let normalized = email.lowercased()
    try sendWelcome(to: normalized)
    return .success
}

// 何時使用 guard let 而不是 if let：
// - 失敗應該退出目前作用域
// - 成功是繼續向下的「快樂路徑」
// 當需要在同一作用域中處理兩個分支時使用 if let。
```

## 總結對比表

| 概念 | JavaScript（Express） | Swift（Vapor） |
|---|---|---|
| 框架搭建 | `npm install express` | 在 `Package.swift` 中新增 Vapor |
| 進入點 | `app.listen(3000)` | `@main` 結構體 + `app.run()` |
| JSON 請求體解析 | `express.json()` 中介軟體 | 透過 `Content` 協定自動處理 |
| 路由定義 | `app.get('/path', handler)` | `app.get("path") { req in }` |
| URL 參數 | `req.params.id` | `req.parameters.require("id")` |
| 請求體 | `req.body.title` | `req.content.decode(DTO.self)` |
| 回應 JSON | `res.json(data)` | 回傳遵循 `Content` 的型別 |
| HTTP 狀態碼 | `res.status(201).json(data)` | `encodeResponse(status: .created)` |
| 錯誤處理 | `next(err)` + 4 參數中介軟體 | `throw Abort(.notFound)` |
| 非同步處理器 | `async (req, res) => { }` | `{ req async throws -> T in }` |
| 中介軟體 | `app.use(fn)` | `app.grouped(MiddlewareType())` |
| 共享狀態 | 模組層級變數 | `actor` + `await` |
| 空值安全 | 可選鏈 `?.` + `??` | `guard let` / `if let` / `??` |

## 後續步驟

你已經完成了 JavaScript → Swift 學習路徑。以下是後續方向：

- **SwiftUI** — 用宣告式 UI 構建原生 iOS/macOS 應用程式（你的下一個 Swift 大冒險）

- **swift.org/server** — 伺服器端 Swift 生態系統、部署指南
- **Hacking with Swift** — hackingwithswift.com，Paul Hudson 的深度教學
- **Swift Forums** — forums.swift.org，語言演進討論的地方
- **WWDC 議程** — developer.apple.com/videos，Apple 的免費深度 Swift 演講
- **Swift Evolution** — github.com/apple/swift-evolution，了解即將到來的特性
