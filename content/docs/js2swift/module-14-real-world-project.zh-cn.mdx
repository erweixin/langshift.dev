## 1. 项目概述

我们将构建 **TaskAPI**——一个包含以下端点的极简任务管理器 REST API：

| 方法 | 路径 | 描述 |
|---|---|---|
| `GET` | `/tasks` | 列出所有任务 |
| `POST` | `/tasks` | 创建任务 |
| `GET` | `/tasks/:id` | 获取单个任务 |
| `PATCH` | `/tasks/:id` | 更新完成状态 |
| `DELETE` | `/tasks/:id` | 删除任务 |

这与你用 Express + 内存存储构建的内容非常接近，因此你可以专注于 Swift/Vapor 的差异。

```swift !! swift
// Swift：Codable 结构体——免费获得序列化！

// Sources/TaskAPI/Models/Task.swift
import Vapor
import Foundation

// Content = Vapor 的请求/响应类型协议（封装了 Codable）
struct Task: Content, Identifiable {
    let id: UUID
    var title: String
    var completed: Bool
    let createdAt: Date

    init(title: String) throws {
        guard !title.trimmingCharacters(in: .whitespaces).isEmpty else {
            throw Abort(.badRequest, reason: "Title cannot be empty")
        }
        self.id = UUID()
        self.title = title.trimmingCharacters(in: .whitespaces)
        self.completed = false
        self.createdAt = Date()
    }
}

// 创建任务的 DTO（只包含客户端发送的内容）
struct CreateTaskRequest: Content {
    let title: String
}

// 更新任务的 DTO
struct UpdateTaskRequest: Content {
    let completed: Bool?
    let title: String?
}

// 内存存储（使用 actor 保证线程安全——见第 9 模块）
actor TaskStore {
    private var tasks: [UUID: Task] = [:]

    func all() -> [Task] { Array(tasks.values).sorted { $0.createdAt < $1.createdAt } }
    func get(_ id: UUID) -> Task? { tasks[id] }
    func create(_ task: Task) { tasks[task.id] = task }
    func update(_ task: Task) { tasks[task.id] = task }
    func delete(_ id: UUID) { tasks.removeValue(forKey: id) }
}
```

## 4. Vapor 中的路由

<UniversalEditor title="路由：Express 与 Vapor 对比" compare={true}>
```javascript !! js
// Express 路由
// src/routes/tasks.js
import { Router } from 'express';
import { Task, taskStore } from '../models/task.js';

const router = Router();

// GET /tasks
router.get('/', (req, res) => {
  const tasks = Array.from(taskStore.values());
  res.json(tasks);
});

// POST /tasks
router.post('/', (req, res) => {
  try {
    const task = new Task(req.body.title);
    taskStore.set(task.id, task);
    res.status(201).json(task);
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});

// GET /tasks/:id
router.get('/:id', (req, res) => {
  const task = taskStore.get(req.params.id);
  if (!task) return res.status(404).json({ error: 'Task not found' });
  res.json(task);
});

// PATCH /tasks/:id
router.patch('/:id', (req, res) => {
  const task = taskStore.get(req.params.id);
  if (!task) return res.status(404).json({ error: 'Task not found' });
  if (req.body.completed !== undefined) task.completed = req.body.completed;
  if (req.body.title !== undefined) task.title = req.body.title;
  taskStore.set(task.id, task);
  res.json(task);
});

// DELETE /tasks/:id
router.delete('/:id', (req, res) => {
  if (!taskStore.delete(req.params.id)) {
    return res.status(404).json({ error: 'Task not found' });
  }
  res.status(204).send();
});

export default router;
```

```swift !! swift
// Vapor 路由
// Sources/TaskAPI/routes.swift
import Vapor

let store = TaskStore()

func routes(_ app: Application) throws {
    let tasks = app.grouped("tasks")

    // GET /tasks
    tasks.get { req async throws -> [Task] in
        await store.all()
        // Vapor 自动将返回的 [Task] 编码为 JSON（它遵循 Content）
    }

    // POST /tasks
    tasks.post { req async throws -> Response in
        let body = try req.content.decode(CreateTaskRequest.self)
        let task = try Task(title: body.title)
        await store.create(task)
        return try await task.encodeResponse(status: .created, for: req)
    }

    // GET /tasks/:id
    tasks.get(":id") { req async throws -> Task in
        let id = try req.parameters.require("id", as: UUID.self)
        guard let task = await store.get(id) else {
            throw Abort(.notFound, reason: "Task \(id) not found")
        }
        return task
    }

    // PATCH /tasks/:id
    tasks.patch(":id") { req async throws -> Task in
        let id = try req.parameters.require("id", as: UUID.self)
        guard var task = await store.get(id) else {
            throw Abort(.notFound, reason: "Task \(id) not found")
        }
        let update = try req.content.decode(UpdateTaskRequest.self)
        if let completed = update.completed { task.completed = completed }
        if let title = update.title { task.title = title }
        await store.update(task)
        return task
    }

    // DELETE /tasks/:id
    tasks.delete(":id") { req async throws -> HTTPStatus in
        let id = try req.parameters.require("id", as: UUID.self)
        guard await store.get(id) != nil else {
            throw Abort(.notFound, reason: "Task \(id) not found")
        }
        await store.delete(id)
        return .noContent   // 204
    }
}
```

```javascript !! js
// Express：异步中间件和路由处理器

import { fetchUserPermissions } from './authService.js';

// 异步中间件
async function requireAuth(req, res, next) {
  try {
    const token = req.headers.authorization?.split(' ')[1];
    if (!token) return res.status(401).json({ error: 'Unauthorized' });
    const permissions = await fetchUserPermissions(token);
    req.user = { permissions };
    next();
  } catch (err) {
    next(err);  // 传递给错误处理器
  }
}

// 异步路由处理器
app.get('/tasks', requireAuth, async (req, res, next) => {
  try {
    const tasks = await db.query('SELECT * FROM tasks');
    res.json(tasks);
  } catch (err) {
    next(err);
  }
});

// 全局错误处理器（必须有 4 个参数）
app.use((err, req, res, next) => {
  console.error(err);
  res.status(err.status ?? 500).json({ error: err.message });
});
```

```swift !! swift
// Vapor：async/await 是默认方式——没有回调地狱

import Vapor

// 异步中间件
struct AuthMiddleware: AsyncMiddleware {
    func respond(to request: Request, chainingTo next: AsyncResponder) async throws -> Response {
        guard let token = request.headers.bearerAuthorization?.token else {
            throw Abort(.unauthorized, reason: "Missing bearer token")
        }
        // 异步调用——直接 await，不需要回调
        let permissions = try await fetchUserPermissions(token: token, db: request.db)
        request.storage[PermissionsKey.self] = permissions
        return try await next.respond(to: request)
    }
}

// 受中间件保护的路由组
func routes(_ app: Application) throws {
    let protected = app.grouped(AuthMiddleware())

    protected.get("tasks") { req async throws -> [Task] in
        // 异步数据库查询——直接 await
        return try await Task.query(on: req.db).all()
    }
}

// 错误处理：从任何地方抛出 Abort 错误
// Vapor 自动捕获并发送正确的 HTTP 响应。
// 自定义错误响应，遵循 AbortError：
struct ValidationError: AbortError {
    var status: HTTPResponseStatus { .unprocessableEntity }
    var reason: String
}
```

```swift !! swift
// Vapor 错误处理——更简洁

// 从路由抛出（路由层面不需要 try/catch）
app.get("tasks", ":id") { req async throws -> Task in

enum TaskError: AbortError {
    case emptyTitle
    case alreadyCompleted(UUID)

    var status: HTTPResponseStatus {
        switch self {
        case .emptyTitle: return .badRequest
        case .alreadyCompleted: return .conflict
        }
    }

    var reason: String {

        case .emptyTitle:
            return "Task title cannot be empty"
        case .alreadyCompleted(let id):
            return "Task \(id) is already completed"
        }
    }
}
```

<UniversalEditor title="常见陷阱 1：滥用强制解包（!）" compare={true}>

```swift !! swift
// Swift 陷阱：过度使用 !（强制解包）

// 错误——如果值为 nil 会在运行时崩溃
let name = user.profile!.name!
let port = Int(ProcessInfo.processInfo.environment["PORT"]!)!

// 正确——使用 guard let 或 if let
guard let portString = ProcessInfo.processInfo.environment["PORT"],
      let port = Int(portString) else {
    fatalError("PORT 环境变量缺失或不是数字")
}

// 正确——使用 nil 合并运算符 ??
let name = user.profile?.name ?? "Anonymous"

// 正确——在函数中使用 guard let 提前退出
func getConfig(_ key: String) throws -> String {
    guard let value = ProcessInfo.processInfo.environment[key],
          !value.isEmpty else {
        throw ConfigError.missing(key: key)
    }
    return value
}

// 规则：只在逻辑上保证值存在时使用 !，
```

```swift !! swift
class TaskViewController {

    var onUpdate: (() -> Void)?

    func setupUpdateHandler() {
        // 错误：`self` 被强引用捕获——循环引用！
```

```swift !! swift
// Swift 陷阱：结构体是值类型——变异创建副本

struct Task {
    var completed: Bool
}

// 错误：函数接收副本；原始值不会被变异
func completeTask(_ task: Task) {
    var copy = task
    copy.completed = true   // 修改副本，不修改原始值
    // 函数外的原始 `task` 保持不变！
}

var myTask = Task(completed: false)
completeTask(myTask)
print(myTask.completed)  // 仍然是 false！

// 正确方案 1：inout 参数
func completeTaskInout(_ task: inout Task) {
    task.completed = true
}
completeTaskInout(&myTask)
print(myTask.completed)  // true

// 正确方案 2：返回修改后的副本
func completed(_ task: Task) -> Task {
    var copy = task
    copy.completed = true
    return copy
}
myTask = completed(myTask)

// 正确方案 3：当变异是主要使用场景时使用类（引用类型）
class TaskRef {
    var completed: Bool = false
}
```

```swift !! swift
// Swift 惯用法：guard let 提前返回（无厄运金字塔）
func processUser(_ user: User?) throws -> ProcessResult {
    // 每个 guard 在失败时提前退出——代码保持扁平
    guard let user = user else {
        throw UserError.noUser
    }
    guard let profile = user.profile else {
        throw UserError.noProfile
    }
    guard let email = profile.email else {
        throw UserError.noEmail
    }
    guard isValidEmail(email) else {
        throw UserError.invalidEmail(email)
    }

    // 到这里，所有值都已安全解包且有效
    let normalized = email.lowercased()
    try sendWelcome(to: normalized)
    return .success
}

// 何时使用 guard let 而不是 if let：
// - 失败应该退出当前作用域
// - 成功是继续向下的"快乐路径"
// 当需要在同一作用域中处理两个分支时使用 if let。
```

```swift !! swift
// Swift：Result<Success, Failure> 用于同步可失败操作
enum FetchError: Error {
    case notFound(UUID)
    case networkError(String)
}

// 同步 Result
func loadCachedTask(id: UUID) -> Result<Task, FetchError> {
    if let task = cache[id] {
        return .success(task)
    }
    return .failure(.notFound(id))
}

// 使用 Result
switch loadCachedTask(id: taskID) {
case .success(let task):
    print("找到：\(task.title)")
case .failure(.notFound(let id)):
    print("没有 id 为 \(id) 的任务")
case .failure(.networkError(let msg)):
    print("网络错误：\(msg)")
}

// 优先使用值类型（结构体）作为数据模型
// 优先使用引用类型（类/actor）用于：
// - 共享可变状态（使用 actor！）
// - 基于标识的对象
// - 继承层次结构

// 惯用的 Swift 数据模型
struct User: Codable {  // 结构体，不是类
    let id: UUID
    var name: String
    var email: String
}

// 共享可变状态——使用 actor，不使用类
actor UserCache {
    private var cache: [UUID: User] = [:]
    func get(_ id: UUID) -> User? { cache[id] }
    func set(_ user: User) { cache[user.id] = user }
}
```

## 总结对比表

| 概念 | JavaScript（Express） | Swift（Vapor） |
|---|---|---|
| 框架搭建 | `npm install express` | 在 `Package.swift` 中添加 Vapor |
| 入口点 | `app.listen(3000)` | `@main` 结构体 + `app.run()` |
| JSON 请求体解析 | `express.json()` 中间件 | 通过 `Content` 协议自动处理 |
| 路由定义 | `app.get('/path', handler)` | `app.get("path") { req in }` |
| URL 参数 | `req.params.id` | `req.parameters.require("id")` |
| 请求体 | `req.body.title` | `req.content.decode(DTO.self)` |
| 响应 JSON | `res.json(data)` | 返回遵循 `Content` 的类型 |
| HTTP 状态码 | `res.status(201).json(data)` | `encodeResponse(status: .created)` |
| 错误处理 | `next(err)` + 4 参数中间件 | `throw Abort(.notFound)` |
| 异步处理器 | `async (req, res) => { }` | `{ req async throws -> T in }` |
| 中间件 | `app.use(fn)` | `app.grouped(MiddlewareType())` |
| 共享状态 | 模块级变量 | `actor` + `await` |
| 空值安全 | 可选链 `?.` + `??` | `guard let` / `if let` / `??` |
